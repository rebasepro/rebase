import { createRebaseClient } from "@rebasepro/client";
import type { TableName, TablesDictionary } from "./generated/sdk/database.types";

export const client = createRebaseClient({
    baseUrl: "",
    tablesDictionary: {},
    offline: { maxCachedRowsPerTable: 100 }
});

export async function list(name: TableName) {
    const rows = await client.data.table(name).find();
    const posts = client.data.postsTable;
    await client.table("posts").find();
    const data = client.data;
    await data.table<{ id: string }>("tags").find();
    return { rows, posts };
}

export function isMissing(error: { code: string }): boolean {
    return error.code === "UNKNOWN_TABLE";
}

export const batch = [{ op: "create", table: "posts", values: {} }];

export type Dictionary = TablesDictionary;

/** A method reference, in a type query. */
export type PostsClient = ReturnType<typeof client.data.table>;

/** An identifier named like Object.prototype's members is nobody's rename. */
export const label = String(client).toString();
