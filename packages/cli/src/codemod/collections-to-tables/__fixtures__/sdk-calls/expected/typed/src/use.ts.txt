import type { CreateRebaseClientResult } from "@rebasepro/client";

/** The project's own structural client: its type is not the SDK, so its calls stay. */
interface Finder { collection(slug: string): { find(): Promise<unknown[]> } }
export async function own(client: { data: Finder }) {
    return client.data.collection("posts").find();
}

/** A receiver the list does not name, recognised by its type. */
export async function count(sdk: CreateRebaseClientResult) {
    const surface = sdk.data;
    return surface.table("posts").count();
}
