import { assertUploadWithinPropertyLimits, UPLOAD_PROPERTY_FIELD, loadCollectionsFromDirectory } from "@rebasepro/server";
import { openRebase } from "@rebasepro/server-postgres";
import { reviewSecurityRules } from "@rebasepro/plugin-review/config";

export const rules = { reviewSecurityRules };

export async function upload(body: Record<string, unknown>) {
    assertUploadWithinPropertyLimits(undefined, { size: 1 });
    const field = body[UPLOAD_PROPERTY_FIELD];
    const rebase = await openRebase({ databaseUrl: process.env.DATABASE_URL, collections: [] });
    return { field, rebase, legacy: loadCollectionsFromDirectory };
}
