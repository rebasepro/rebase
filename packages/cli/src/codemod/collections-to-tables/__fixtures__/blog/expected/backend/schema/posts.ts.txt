/**
 * Blog posts: the public site reads them, editors write them.
 */

import { defineTable, col, rel } from "@rebasepro/server/schema";
import { tags } from "./tags";
import { authors } from "./authors";
import { slugify } from "../lib/slugify";

export const posts = defineTable("posts", {
    columns: {
        id: col.uuid().primaryKey().defaultRandom(),
        // Shown as the page heading.
        title: col.text().notNull(),
        slug: col.text().unique(),
        status: col.enum(["draft", "published"]).default("draft"),
        body: col.text(),
        cover: col.storage({ path: "posts/covers/", accept: ["image/*"] }),
        authorId: col.uuid(),
        __order: col.text()
    },
    relations: {
        tags: rel.manyToMany(() => tags, { through: "posts_tags" }),
        author: rel.belongsTo(() => authors, { onDelete: "set null" })
    },
    rls: "inherit",
    hooks: {
        afterRead: ({ values: row }) => {
            return row;
        },
        beforeSave: ({ values, previousValues }) => {
            const merged = { ...previousValues, ...values };
            if (typeof merged.title === "string" && !merged.slug) values.slug = slugify(merged.title);
            return values;
        }
    }
});
