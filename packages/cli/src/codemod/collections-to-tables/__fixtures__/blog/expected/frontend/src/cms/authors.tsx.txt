import { customizeTable } from "@rebasepro/cms";
import type { authors } from "../../../backend/schema/authors";

export const authorsCms = customizeTable<typeof authors>("authors", {
    group: "Content",
    fields: {
        email: {
            name: "Email address"
        },
        avatar: {
            storage: {
                path: ({ values, entityId }) => {
                    const owner = typeof values?.handle === "string" && values.handle ? values.handle : `authors/${entityId}`;
                    return `${owner}/avatar`;
                }
            }
        }
    }
});
