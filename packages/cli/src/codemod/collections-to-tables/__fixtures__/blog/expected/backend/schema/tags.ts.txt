import { defineTable, col } from "@rebasepro/server/schema";
import { tagCallbacks } from "../../config/tag-callbacks";

/** Nothing here is presentation the CMS would not derive: no customization file. */
export const tags = defineTable("tags", {
    columns: {
        id: col.integer().primaryKey().generatedByDefaultAsIdentity(),
        name: col.text().notNull()
    },
    // Authors read the tags; built with 0.30's policy.existsIn({ collection }).
    rls: [
        {
            operation: "select",
            condition: {
                kind: "existsIn",
                table: "authors",
                where: { kind: "compare", op: "eq", left: { kind: "field", name: "id" }, right: { kind: "authUid" } }
            }
        }
    ],
    hooks: tagCallbacks
});
