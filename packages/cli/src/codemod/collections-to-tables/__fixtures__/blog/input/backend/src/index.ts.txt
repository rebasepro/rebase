import path from "node:path";
import { initializeRebaseBackend } from "@rebasepro/server";
import usersCollection from "../../config/collections/users";
import { storageAuthorize } from "../../config/storage";

// A custom backend entry: the collections directory, the auth collection, the storage rule.
await initializeRebaseBackend({
    collectionsDir: path.join(__dirname, "../../config/collections"),
    auth: {
        collection: usersCollection,
        jwtSecret: process.env.JWT_SECRET ?? ""
    },
    storageAuthorize
});
