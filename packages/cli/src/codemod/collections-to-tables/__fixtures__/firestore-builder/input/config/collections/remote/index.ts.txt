import type { FirebaseCollectionConfig } from "@rebasepro/types";
import visitsCollection from "./visits";
import { getPagesCollection } from "./pages";

export const locales = ["en", "de"];

/** The localized pages, one collection per locale. */
export function buildPageCollections(): FirebaseCollectionConfig[] {
    return locales.map(locale => getPagesCollection(locale));
}

/** Gives every collection a read-only `id` field, at run time. */
function withIdFields(collections: FirebaseCollectionConfig[]): FirebaseCollectionConfig[] {
    return collections.map(collection => ({ ...collection, properties: { id: { name: "ID", type: "string", readOnly: true }, ...collection.properties } }));
}

/** Every Firestore collection; the app spreads them into the panel's collections. */
export function getRemoteCollections(): FirebaseCollectionConfig[] {
    return withIdFields([visitsCollection, ...buildPageCollections()]);
}
